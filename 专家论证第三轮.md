> 目的：契约未声明 `useAiChat` 签名。本文件把假设集中到一处，若第 4 轮审计发现签名不同，只改此文件即可，业务页面不动。不涉及 SSE 解析（契约禁止自研流解析）。

```ts
import { useAiChat } from './useAiChat';

export interface AiChatCompat {
  content: string;
  isLoading: boolean;
  error: string | null;
  send: (prompt: string) => void;
  reset: () => void;
}

type RawAiChat = {
  content?: string;
  text?: string;
  isLoading?: boolean;
  loading?: boolean;
  error?: string | null;
  send?: (p: string) => void;
  stream?: (p: string) => void;
  run?: (p: string) => void;
  reset?: () => void;
  clear?: () => void;
} | null | undefined;

export function useAiChatCompat(): AiChatCompat {
  const raw = useAiChat() as RawAiChat;
  return {
    content: raw?.content ?? raw?.text ?? '',
    isLoading: raw?.isLoading ?? raw?.loading ?? false,
    error: raw?.error ?? null,
    send: (p: string) => {
      const fn = raw?.send ?? raw?.stream ?? raw?.run;
      if (typeof fn === 'function') fn(p);
    },
    reset: () => {
      const fn = raw?.reset ?? raw?.clear;
      if (typeof fn === 'function') fn();
    },
  };
}
```

```tsx
import type { ReactNode } from 'react';

export type Confidence = 'verified' | 'probable' | 'disputed' | 'legendary';

export interface EvidenceItem {
  id: string;
  label: string;
  value?: string;
  source?: string;
  note?: string;
  confidence: Confidence;
}

export interface FortuneEvidenceCardProps {
  title?: string;
  items: EvidenceItem[];
  emptyText?: string;
  className?: string;
  footer?: ReactNode;
}

const CONF_LABEL: Record<Confidence, string> = {
  verified: '已验证',
  probable: '较可信',
  disputed: '有争议',
  legendary: '文化习俗',
};

const CONF_ORDER: Confidence[] = ['verified', 'probable', 'disputed', 'legendary'];

function genId(): string {
  const c = (globalThis as { crypto?: { randomUUID?: () => string } }).crypto;
  if (c?.randomUUID) return c.randomUUID();
  return `e_${Math.random().toString(36).slice(2, 10)}`;
}

function normalizeConfidence(v: unknown): Confidence {
  return v === 'verified' || v === 'probable' || v === 'disputed' || v === 'legendary'
    ? v
    : 'legendary';
}

/** 把未知形状的 API 证据数组规范化为 EvidenceItem[]；非数组返回 [] */
export function normalizeEvidence(raw: unknown): EvidenceItem[] {
  if (!Array.isArray(raw)) return [];
  const out: EvidenceItem[] = [];
  for (let i = 0; i < raw.length; i++) {
    const r = raw[i];
    if (!r || typeof r !== 'object') continue;
    const o = r as Record<string, unknown>;
    const label = typeof o.label === 'string' && o.label.trim() ? o.label : `证据 ${i + 1}`;
    out.push({
      id: typeof o.id === 'string' && o.id ? o.id : genId(),
      label,
      value: typeof o.value === 'string' ? o.value : undefined,
      source: typeof o.source === 'string' ? o.source : undefined,
      note: typeof o.note === 'string' ? o.note : undefined,
      confidence: normalizeConfidence(o.confidence),
    });
  }
  return out;
}

export function FortuneEvidenceCard(props: FortuneEvidenceCardProps) {
  const { title = '证据', items, emptyText = '暂无证据数据', className, footer } = props;
  const safeItems = Array.isArray(items) ? items : [];
  const grouped = CONF_ORDER
    .map(c => ({ c, list: safeItems.filter(i => i.confidence === c) }))
    .filter(g => g.list.length > 0);

  return (
    <section className={`ts-evidence-card ${className ?? ''}`.trim()} aria-label={title}>
      <header className="ts-evidence-card__head">
        <h3 className="ts-evidence-card__title">{title}</h3>
        <span className="ts-evidence-card__count">{safeItems.length} 条</span>
      </header>
      {safeItems.length === 0 && <div className="ts-empty ts-evidence-card__empty">{emptyText}</div>}
      {grouped.map(g => (
        <div key={g.c} className={`ts-evidence-card__group ts-evidence-card__group--${g.c}`}>
          <div className={`ts-badge ts-badge--${g.c}`}>{CONF_LABEL[g.c]}</div>
          <ul className="ts-evidence-card__list">
            {g.list.map(it => (
              <li key={it.id} className="ts-evidence-card__item">
                <div className="ts-evidence-card__label">{it.label}</div>
                {it.value && <div className="ts-evidence-card__value">{it.value}</div>}
                {(it.source || it.note) && (
                  <div className="ts-evidence-card__meta">
                    {it.source && <span>出处：{it.source}</span>}
                    {it.note && <span>口径：{it.note}</span>}
                  </div>
                )}
              </li>
            ))}
          </ul>
        </div>
      ))}
      {footer && <footer className="ts-evidence-card__footer">{footer}</footer>}
    </section>
  );
}

export default FortuneEvidenceCard;
```

```typescript
import { useState, useEffect } from 'react';
import { safeStorage } from '../lib/safe-storage'; // 契约 §6 指定存储库

export interface AlmanacRhythm {
  current_shichen: string;
  tcm_meridian: string;
  action_advice: string;
  compliance_tag: string;
}

export interface AlmanacYiJi {
  yi: string[];
  ji: string[];
  evidence: string;
}

export interface AlmanacAstroOverlay {
  moon_phase: string;
  major_aspect: string;
}

export interface AlmanacData {
  date: string;
  lunar: string;
  ganzhi: { year: string; month: string; day: string };
  rhythm: AlmanacRhythm;
  yi_ji: AlmanacYiJi;
  astro_overlay: AlmanacAstroOverlay;
  disclaimer: string;
}

interface UseAlmanacDataReturn {
  data: AlmanacData | null;
  loading: boolean;
  error: string | null;
}

// 获取设备本地时区的 YYYY-MM-DD 格式日期，防止跨日错乱
function getLocalDateString(date: Date = new Date()): string {
  const formatter = new Intl.DateTimeFormat('en-CA', {
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
    timeZone: Intl.DateTimeFormat().resolvedOptions().timeZone
  });
  return formatter.format(date);
}

export function useAlmanacData(): UseAlmanacDataReturn {
  const [data, setData] = useState<AlmanacData | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const fetchAlmanac = async () => {
      setLoading(true);
      setError(null);
      const localDate = getLocalDateString();

      // 契约 §4 强制要求：safe-storage 持久化，键前缀 temposoul:fortune:
      const cacheKey = `temposoul:fortune:almanac:${localDate}`;
      const cached = safeStorage.getItem(cacheKey);
      if (cached) {
        try {
          setData(JSON.parse(cached));
          setLoading(false);
          return;
        } catch (e) {
          safeStorage.removeItem(cacheKey);
        }
      }

      try {
        const response = await fetch('/api/v1/divination/almanac', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            dateType: 'solar',
            startDate: localDate,
            endDate: localDate,
            responseMode: 'full'
          })
        });

        const json = await response.json();
        if (!json.ok) {
          throw new Error(json.error?.message || '黄历数据加载失败');
        }

        const mappedData: AlmanacData = {
          date: localDate,
          lunar: json.data.lunar || '未知',
          ganzhi: json.data.ganzhi || { year: '', month: '', day: '' },
          rhythm: json.data.rhythm || {
            current_shichen: '未知',
            tcm_meridian: '未知',
            action_advice: '暂无节律建议',
            compliance_tag: '传统文化科普，非医疗建议'
          },
          yi_ji: json.data.yi_ji || { yi: [], ji: [], evidence: '' },
          astro_overlay: json.data.astro_overlay || { moon_phase: '', major_aspect: '' },
          disclaimer: '本内容基于传统历法与天文算法生成，仅供文化参考。'
        };

        setData(mappedData);
        safeStorage.setItem(cacheKey, JSON.stringify(mappedData));
      } catch (err: any) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };

    fetchAlmanac();
  }, []);

  return { data, loading, error };
}
```

```
export function stripTone(p: string): string {
  return p.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase();
}

/** 汉字一字一音节，直接以每字拼音作为一个音节 */
export function splitSyllables(pinyinList: string[]): string[] {
  return pinyinList.filter((p) => Boolean(p && p.trim()));
}

export function isTongueTwister(pinyinList: string[]): boolean {
  const norm = pinyinList.map(stripTone);
  for (let i = 1; i < norm.length; i++) {
    const a = norm[i - 1], b = norm[i];
    if (!a || !b) continue;
    if (a === b) return true;
    const vA = a.replace(/^[^aeiouüv]+/, ''), vB = b.replace(/^[^aeiouüv]+/, '');
    if (vA && vA === vB && a[0] === b[0]) return true;
  }
  return false;
}
```

```
export const STRONG_ASSERTIONS = ['大吉','大凶','吉凶','注定','必然','百分百','成功率','必'];
export const WEAK_ASSERTIONS   = ['旺','克','一定','吉','凶'];
export const ALLOW_LIST        = ['克服','克己','吉利','吉时','旺盛'];
const MARK = '［断言已过滤］';
const BOUNDARY = '(?=[，。！？；、\\s]|$)';

export function guardText(text: string): string {
  if (!text) return '';
  let out = text;
  for (const w of STRONG_ASSERTIONS) {
    if (ALLOW_LIST.some((a) => out.includes(a))) continue;
    out = out.split(w).join(MARK);            // 强词：全局替换
  }
  for (const w of WEAK_ASSERTIONS) {
    if (ALLOW_LIST.some((a) => out.includes(a))) continue;
    out = out.replace(new RegExp(`${w}${BOUNDARY}`, 'g'), MARK); // 弱词：仅边界
  }
  return out;
}
```

## 1. D-P0-1: 全局搜索入口组件 `SearchInput`

### ① 功能方案 (PRD 片段)

- **功能名**：全局搜索入口 (`SearchInput`)
- **输入字段**：搜索关键词 (string, max 100 chars)
- **输出结构**：无直接渲染输出，触发路由跳转至 `/search?q={encoded_query}`
- **交互**：输入框支持防抖（由表单提交自然防抖），支持 `Cmd/Ctrl + K` 全局快捷键聚焦。
- **边界**：空输入或纯空格不触发跳转；特殊字符通过 `encodeURIComponent` 安全转义，杜绝 XSS 注入风险。
- **工作量**：0.5 天

### ② 完整代码

**文件路径**: `src/components/search/SearchInput.tsx`

```tsx
import React, { useState, useEffect, useRef } from 'react';
import { useNavigate } from 'react-router-dom';

export const SearchInput: React.FC = () => {
  const [query, setQuery] = useState('');
  const navigate = useNavigate();
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
        e.preventDefault();
        inputRef.current?.focus();
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, []);

  const handleSearch = (e: React.FormEvent) => {
    e.preventDefault();
    const trimmedQuery = query.trim();
    if (trimmedQuery) {
      // 安全转义，防止 URL 注入
      navigate(`/search?q=${encodeURIComponent(trimmedQuery)}`);
    }
  };

  return (
    <form onSubmit={handleSearch} className="search-input__form" role="search">
      <input
        ref={inputRef}
        type="text"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        placeholder="搜索词库、功能..."
        className="search-input__field"
        aria-label="全局搜索"
        maxLength={100}
      />
      <kbd className="search-input__shortcut">⌘K</kbd>
    </form>
  );
};
```

## 2. D-P0-2: 搜索聚合结果页 `/search`

### ① 功能方案 (PRD 片段)

- **功能名**：搜索聚合结果页 (`SearchPage`)
- **输入字段**：URL Query 参数 `?q=`
- **输出结构**：按 Tab (词库/功能入口/资讯) 分类的匹配列表，严格映射 `SearchIndexItem` Schema。
- **交互**：读取 URL 参数，使用 `fuse.js` 在客户端对静态索引进行模糊匹配；点击结果项跳转至对应 `url`。
- **边界**：无查询参数时渲染空状态；未匹配时渲染 `ZeroStateFallback`；资讯 Tab 当前显示“P1 预留”占位，不请求任何后端 API。
- **工作量**：1.5 天

### ② 完整代码

**文件路径**: `src/pages/platform/SearchPage.tsx`

```tsx
import React, { useMemo, useState, useEffect } from 'react';
import { useSearchParams, Link } from 'react-router-dom';
import Fuse from 'fuse.js';
import { lexicon } from '../../data/lexicon'; 
import { ZeroStateFallback } from '../../components/search/ZeroStateFallback';
import { trackSearch } from '../../lib/analytics';

// 严格遵循 SearchIndexItem Schema (v3.2 §5)
interface FeatureEntry {
  id: string;
  type: 'feature';
  title: string;
  summary: string;
  tags: string[];
  url: string;
  lang: string;
  confidence: 'verified' | 'probable' | 'disputed' | 'legendary';
  updatedAt: string;
}

const FEATURE_ENTRIES: FeatureEntry[] = [
  { 
    id: 'f1', type: 'feature', title: '八字排盘', 
    summary: '专业八字大运流年排盘与十神分析', 
    tags: ['八字', '排盘'], url: '/?mode=single&system=bazi', 
    lang: 'zh-CN', confidence: 'verified', updatedAt: '2026-09-16' 
  },
  { 
    id: 'f2', type: 'feature', title: '紫微斗数', 
    summary: '紫微十二宫与四化飞星深度解析', 
    tags: ['紫微', '排盘'], url: '/?mode=single&system=ziwei', 
    lang: 'zh-CN', confidence: 'verified', updatedAt: '2026-09-16' 
  },
  { 
    id: 'f3', type: 'feature', title: '姓名测试', 
    summary: '五格三才与生肖音形义综合打分', 
    tags: ['姓名', '测试'], url: '/name-test', 
    lang: 'zh-CN', confidence: 'verified', updatedAt: '2026-09-16' 
  },
];

// 合并静态数据源构建客户端搜索索引，防御性处理 lexicon 可能为空的情况
const SEARCH_INDEX = [
  ...(Array.isArray(lexicon) ? lexicon : []).map((item: any) => ({ ...item, type: 'lexicon' as const })),
  ...FEATURE_ENTRIES
];

type TabType = 'lexicon' | 'feature' | 'article';

export const SearchPage: React.FC = () => {
  const [searchParams] = useSearchParams();
  const query = searchParams.get('q') || '';
  const [activeTab, setActiveTab] = useState<TabType>('lexicon');

  useEffect(() => {
    if (query) {
      trackSearch(query);
    }
  }, [query]);

  const fuse = useMemo(() => new Fuse(SEARCH_INDEX, {
    keys: ['title', 'summary', 'tags'],
    threshold: 0.3,
    includeScore: true,
  }), []);

  const results = useMemo(() => {
    if (!query.trim()) return [];
    return fuse.search(query).map(result => result.item);
  }, [query, fuse]);

  const filteredResults = useMemo(() => {
    if (activeTab === 'article') return []; // P1 预留，不发起任何网络请求
    return results.filter(r => r.type === activeTab);
  }, [results, activeTab]);

  if (!query.trim()) {
    return <ZeroStateFallback query="" />;
  }

  return (
    <div className="search-page">
      <header className="search-page__header">
        <h1 className="search-page__title">搜索结果: "{query}"</h1>
        <div className="search-page__tabs" role="tablist">
          {(['lexicon', 'feature', 'article'] as TabType[]).map(tab => (
            <button
              key={tab}
              role="tab"
              aria-selected={activeTab === tab}
              className={`search-page__tab ${activeTab === tab ? 'search-page__tab--active' : ''}`}
              onClick={() => setActiveTab(tab)}
            >
              {tab === 'lexicon' ? '词库' : tab === 'feature' ? '功能入口' : '资讯 (P1)'}
            </button>
          ))}
        </div>
      </header>

      <main className="search-page__results">
        {filteredResults.length === 0 ? (
          <ZeroStateFallback query={query} />
        ) : (
          <ul className="search-page__list">
            {filteredResults.map((item) => (
              <li key={item.id} className="search-result-item">
                <Link to={item.url} className="search-result-item__link">
                  <span className="search-result-item__type-tag">{item.type}</span>
                  <h3 className="search-result-item__title">{item.title}</h3>
                  <p className="search-result-item__summary">{item.summary}</p>
                  {item.confidence && (
                    <span className={`search-result-item__confidence confidence--${item.confidence}`}>
                      置信度: {item.confidence}
                    </span>
                  )}
                </Link>
              </li>
            ))}
          </ul>
        )}
      </main>
    </div>
  );
};
```
